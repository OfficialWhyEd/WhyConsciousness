def aggiorna_neurochimica(personaggio):

def prendi_decisione(personaggio, opzioni):
    punteggi = []
    for opzione in opzioni:
        punteggio = (
            opzione.utilità * 0.5 + 
            opzione.impatto_sociale * personaggio.ossitocina +

def dialogo(personaggio_A, personaggio_B):

        nuova_credenza = opinione_A.fusione(opinione_B)
